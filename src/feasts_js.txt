function easter(y){const a=y%19,b=Math.floor(y/100),c=y%100,d=Math.floor(b/4),e=b%4,f=Math.floor((b+8)/25),g=Math.floor((b-f+1)/3),
 h=(19*a+b-d-g+15)%30,i=Math.floor(c/4),k=c%4,l=(32+2*e+2*i-h-k)%7,m=Math.floor((a+11*h+22*l)/451),
 mo=Math.floor((h+l-7*m+114)/31),da=((h+l-7*m+114)%31)+1;return Dt(y,mo,da)}
function advent1(y){const c=Dt(y,12,25),w=c.getDay()||7;return addD(c,-w-21)}   // 4. niedziela przed 25.12
function feastOf(d){
 const y=d.getFullYear(),n=dayN(d),m=d.getMonth()+1,dd=d.getDate(),E=dayN(easter(y)),A1=dayN(advent1(y)),r=n-E;
 if((m===12&&dd>=24)||(m===1&&dd<=6))return {id:'boze',name:m===1&&dd===6?'Epifania':m===12&&dd===24?'Wigilia':'Boże Narodzenie'};
 if(n>=A1&&m>=11){const c=Math.min(4,Math.floor((n-A1)/7)+1);return {id:'adwent',candles:c,name:c+'. tydzień adwentu'}}
 if(n===A1-7)return {id:'wiecznosc',name:'Niedziela Wieczności'};
 if(m===10&&dd===31)return {id:'reformacja',name:'Święto Reformacji'};
 if(m===11&&dd===1)return {id:'zmarli',name:'Dzień Pamięci Zmarłych'};
 if(HARVEST.includes(iso(d)))return {id:'zniwa',name:'Święto Żniw'};
 if(r===-7)return {id:'palmowa',name:'Niedziela Palmowa',pasja:true};
 if(r===-3)return {id:'wczwartek',name:'Wielki Czwartek'};
 if(r===-2)return {id:'wpiatek',name:'Wielki Piątek'};
 if(r===-1)return {id:'wsobota',name:'Wielka Sobota'};
 if(r===0||r===1)return {id:'wielkanoc',name:r?'Poniedziałek Wielkanocny':'Wielkanoc'};
 if(r>=-46&&r<=-4)return {id:'pasja',name:r===-46?'Dzień Pokuty i Modlitwy':'Okres pasyjny',pasja:true};
 if(r===39)return {id:'wniebo',name:'Wniebowstąpienie'};
 if(r===49||r===50)return {id:'zielone',name:r===49?'Zesłanie Ducha Świętego':'Zesłanie Ducha Świętego – drugie święto'};
 return null}
